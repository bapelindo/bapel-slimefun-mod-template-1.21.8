package com.example.automation.impl;  

import com.example.automation.infra.NetworkGridLocator;  
import net.minecraft.block.entity.BlockEntity;  
import net.minecraft.client.MinecraftClient;  
import net.minecraft.util.math.BlockPos;  
import net.minecraft.world.World;  

import java.util.Comparator;  
import java.util.Optional;  

public class NetworkGridLocatorImpl implements NetworkGridLocator {  

    // ID blok Slimefun Network Grid — sesuaikan dengan mod kamu  
    private static final String GRID_BLOCK_ID = "slimefun:network_grid";  

    @Override  
    public Optional<BlockPos> findNearest(BlockPos origin, int radius) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        World world = mc.world;  
        if (world == null) return Optional.empty();  

        return BlockPos.streamOutwards(origin, radius, radius, radius)  
            .filter(pos -> isGridActive(pos))  
            .min(Comparator.comparingDouble(pos -> pos.getSquaredDistance(origin)))  
            .map(BlockPos::toImmutable);  
    }  

    @Override  
    public boolean isGridActive(BlockPos pos) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (mc.world == null) return false;  

        // Cek block ID  
        String blockId = mc.world.getBlockState(pos)  
            .getBlock()  
            .getTranslationKey();  

        if (!blockId.contains("network_grid")) return false;  

        // Cek BlockEntity untuk status aktif  
        BlockEntity be = mc.world.getBlockEntity(pos);  
        return be != null; // extend: cek NBT "active" tag  
    }  
}

package com.example.automation.impl;  

import com.example.automation.infra.FastMachineLocator;  
import net.minecraft.client.MinecraftClient;  
import net.minecraft.item.ItemStack;  
import net.minecraft.util.math.BlockPos;  
import net.minecraft.world.World;  

import java.util.Comparator;  
import java.util.Optional;  
import java.util.Set;  

public class FastMachineLocatorImpl implements FastMachineLocator {  

    // Set semua ID mesin yang dikenal — configurable  
    private static final Set<String> KNOWN_MACHINE_IDS = Set.of(  
        "slimefun:electric_smeltery",  
        "slimefun:electric_ingot_factory",  
        "slimefun:electric_ore_grinder",  
        "slimefun:electric_gold_pan",  
        "bapelmod:fast_machine"  
    );  

    @Override  
    public Optional<BlockPos> findNearest(BlockPos origin, ItemStack targetItem, int radius) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        World world = mc.world;  
        if (world == null) return Optional.empty();  

        return BlockPos.streamOutwards(origin, radius, radius, radius)  
            .filter(pos -> isMachineAvailable(pos) && supportsRecipe(pos, targetItem))  
            .min(Comparator.comparingDouble(pos -> pos.getSquaredDistance(origin)))  
            .map(BlockPos::toImmutable);  
    }  

    @Override  
    public boolean isMachineAvailable(BlockPos pos) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (mc.world == null) return false;  

        String blockId = mc.world.getBlockState(pos)  
            .getBlock()  
            .getTranslationKey();  

        return KNOWN_MACHINE_IDS.stream().anyMatch(blockId::contains);  
    }  

    @Override  
    public boolean supportsRecipe(BlockPos pos, ItemStack targetItem) {  
        // TODO: Query SlimefunRecipeRegistry untuk cek apakah  
        // mesin di pos ini support recipe untuk targetItem  
        // Untuk sekarang return true jika mesin tersedia  
        return isMachineAvailable(pos);  
    }  
}  
CameraControllerImpl.java
java
package com.example.automation.impl;  

import com.example.automation.infra.CameraController;  
import net.minecraft.client.MinecraftClient;  
import net.minecraft.entity.player.PlayerEntity;  
import net.minecraft.util.math.BlockPos;  
import net.minecraft.util.math.MathHelper;  
import net.minecraft.util.math.Vec3d;  

public class CameraControllerImpl implements CameraController {  

    private static final float LOOK_SPEED      = 5.0f;   // derajat per tick  
    private static final float LOOK_TOLERANCE  = 2.0f;   // derajat toleransi  
    private static final float EYE_HEIGHT      = 1.62f;  

    private BlockPos currentTarget = null;  
    private boolean  controlled    = false;  

    @Override  
    public void lookAt(BlockPos target) {  
        this.currentTarget = target;  
        this.controlled    = true;  
        tickLook(); // mulai rotate sekarang  
    }  

    @Override  
    public boolean isLookingAt(BlockPos target) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        PlayerEntity player = mc.player;  
        if (player == null) return false;  

        float[] desired = calculateAngles(player, target);  
        float   dyaw    = Math.abs(MathHelper.wrapDegrees(player.getYaw()   - desired[0]));  
        float   dpitch  = Math.abs(MathHelper.wrapDegrees(player.getPitch() - desired[1]));  

        boolean looking = dyaw < LOOK_TOLERANCE && dpitch < LOOK_TOLERANCE;  

        // Terus rotate setiap tick sampai on-target  
        if (!looking && controlled) tickLook();  

        return looking;  
    }  

    @Override  
    public void release() {  
        this.currentTarget = null;  
        this.controlled    = false;  
    }  

    @Override  
    public boolean isControlled() { return controlled; }  

    // ─────────────────────────────────────────────────────────────  

    private void tickLook() {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        PlayerEntity player = mc.player;  
        if (player == null || currentTarget == null) return;  

        float[] desired = calculateAngles(player, currentTarget);  

        float currentYaw   = player.getYaw();  
        float currentPitch = player.getPitch();  

        float newYaw   = lerpAngle(currentYaw,   desired[0], LOOK_SPEED);  
        float newPitch = lerpAngle(currentPitch, desired[1], LOOK_SPEED);  

        player.setYaw(newYaw);  
        player.setPitch(newPitch);  
    }  

    private float[] calculateAngles(PlayerEntity player, BlockPos target) {  
        Vec3d eye    = player.getPos().add(0, EYE_HEIGHT, 0);  
        Vec3d center = Vec3d.ofCenter(target);  
        Vec3d delta  = center.subtract(eye);  

        double dist = Math.sqrt(delta.x * delta.x + delta.z * delta.z);  
        float  yaw  = (float) Math.toDegrees(Math.atan2(-delta.x, delta.z));  
        float  pitch= (float) Math.toDegrees(-Math.atan2(delta.y, dist));  

        return new float[]{ yaw, pitch };  
    }  

    private float lerpAngle(float current, float target, float speed) {  
        float diff = MathHelper.wrapDegrees(target - current);  
        float step = MathHelper.clamp(diff, -speed, speed);  
        return current + step;  
    }  
}  
InventoryCheckerImpl.java

package com.example.automation.impl;  

import com.example.automation.infra.InventoryChecker;  
import net.minecraft.client.MinecraftClient;  
import net.minecraft.item.ItemStack;  
import net.minecraft.nbt.NbtCompound;  

public class InventoryCheckerImpl implements InventoryChecker {  

    @Override  
    public int countInInventory(ItemStack target) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (mc.player == null) return 0;  

        String targetKey = getSlimefunKey(target);  
        int total = 0;  

        for (ItemStack stack : mc.player.getInventory().main) {  
            if (stack.isEmpty()) continue;  

            if (targetKey != null) {  
                // Bandingkan via Slimefun ID (NBT-based)  
                if (targetKey.equals(getSlimefunKey(stack))) {  
                    total += stack.getCount();  
                }  
            } else {  
                // Fallback: vanilla item match  
                if (ItemStack.areItemsEqual(stack, target)) {  
                    total += stack.getCount();  
                }  
            }  
        }  

        return total;  
    }  

    @Override  
    public boolean hasInventorySpace() {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (mc.player == null) return false;  

        return mc.player.getInventory().main.stream()  
            .anyMatch(ItemStack::isEmpty);  
    }  

    private String getSlimefunKey(ItemStack stack) {  
        if (stack.isEmpty()) return null;  
        NbtCompound nbt = stack.getNbt();  
        if (nbt == null) return null;  
        // Slimefun stores ID di: PublicBukkitValues → slimefun:slimefun_item  
        // Di client-side NBT Compound:  
        NbtCompound sfTag = nbt.getCompound("PublicBukkitValues");  
        if (sfTag.isEmpty()) return null;  
        return sfTag.getString("slimefun:slimefun_item");  
    }  
}

package com.example.automation.impl;  

import com.example.automation.infra.GuiInteractor;  
import net.minecraft.client.MinecraftClient;  
import net.minecraft.client.gui.screen.ingame.HandledScreen;  
import net.minecraft.network.packet.c2s.play.PlayerInteractBlockC2SPacket;  
import net.minecraft.util.Hand;  
import net.minecraft.util.hit.BlockHitResult;  
import net.minecraft.util.math.BlockPos;  
import net.minecraft.util.math.Direction;  
import net.minecraft.util.math.Vec3d;  

public class GuiInteractorImpl implements GuiInteractor {  

    private static final String GRID_GUI_KEYWORD    = "network_grid";  
    private static final String MACHINE_GUI_KEYWORD = "fast_machine";  

    @Override  
    public void rightClickBlock(BlockPos pos) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (mc.player == null || mc.getNetworkHandler() == null) return;  

        BlockHitResult hitResult = new BlockHitResult(  
            Vec3d.ofCenter(pos),  
            Direction.NORTH,  
            pos,  
            false  
        );  

        mc.getNetworkHandler().sendPacket(  
            new PlayerInteractBlockC2SPacket(Hand.MAIN_HAND, hitResult, 0)  
        );  
    }  

    @Override  
    public void closeCurrentGui() {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (mc.currentScreen != null) {  
            mc.player.closeHandledScreen();  
            mc.setScreen(null);  
        }  
    }  

    @Override  
    public boolean isGridGuiOpen() {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (!(mc.currentScreen instanceof HandledScreen<?> screen)) return false;  
        return screen.getTitle().getString().toLowerCase()  
            .contains(GRID_GUI_KEYWORD);  
    }  

    @Override  
    public boolean isMachineGuiOpen() {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (!(mc.currentScreen instanceof HandledScreen<?> screen)) return false;  
        return screen.getTitle().getString().toLowerCase()  
            .contains(MACHINE_GUI_KEYWORD);  
    }  

    @Override  
    public void shiftClickSlot(int slotIndex) {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (!(mc.currentScreen instanceof HandledScreen<?> screen)) return;  

        // Kirim shift-click packet ke slot  
        mc.interactionManager.clickSlot(  
            screen.getScreenHandler().syncId,  
            slotIndex,  
            0,   // button = left click  
            net.minecraft.screen.slot.SlotActionType.QUICK_MOVE,  
            mc.player  
        );  
    }  

    @Override  
    public int getActiveGuiSlotCount() {  
        MinecraftClient mc = MinecraftClient.getInstance();  
        if (!(mc.currentScreen instanceof HandledScreen<?> screen)) return 0;  
        return screen.getScreenHandler().slots.size();  
    }  
}

package com.example.automation.hud;  

import com.example.automation.AutomationContext;  
import com.example.automation.AutomationStateMachine;  
import com.example.automation.CraftingJob;  
import com.example.automation.event.AutomationEvent;  
import com.example.automation.event.AutomationEventBus;  
import com.example.automation.state.AutomationState;  
import net.fabricmc.fabric.api.client.rendering.v1.HudRenderCallback;  
import net.minecraft.client.MinecraftClient;  
import net.minecraft.client.font.TextRenderer;  
import net.minecraft.client.gui.DrawContext;  

import java.util.ArrayList;  
import java.util.List;  

/**  
 * Renders automation status overlay di pojok kiri atas layar.  
 *  
 * Registrasikan di ClientModInitializer:  
 *   HudRenderCallback.EVENT.register(renderer::render);  
 */  
public class AutomationHudRenderer {  

    // ── Colors ────────────────────────────────────────────────────  
    private static final int COLOR_BG          = 0xAA000000;  
    private static final int COLOR_TITLE        = 0xFFFFD700; // gold  
    private static final int COLOR_STATE_ACTIVE = 0xFF55FF55; // green  
    private static final int COLOR_STATE_WAIT   = 0xFFFFAA00; // orange  
    private static final int COLOR_STATE_ERROR  = 0xFFFF5555; // red  
    private static final int COLOR_JOB          = 0xFFAAAAAA; // gray  
    private static final int COLOR_DEPTH        = 0xFF5555FF; // blue  

    // ── Layout ────────────────────────────────────────────────────  
    private static final int PADDING  = 4;  
    private static final int LINE_H   = 10;  
    private static final int X_OFFSET = 4;  
    private static final int Y_OFFSET = 4;  

    private final AutomationStateMachine stateMachine;  
    private final AutomationContext      ctx;  

    // Live data dari EventBus  
    private volatile String lastError  = null;  
    private volatile long   errorTimer = 0L;  

    public AutomationHudRenderer(AutomationStateMachine stateMachine, AutomationContext ctx) {  
        this.stateMachine = stateMachine;  
        this.ctx          = ctx;  

        // Subscribe error events untuk ditampilkan di HUD  
        AutomationEventBus.get().subscribe(AutomationEvent.CHAIN_FAILED, reason -> {  
            lastError  = (String) reason;  
            errorTimer = System.currentTimeMillis();  
        });  
    }  

    // ─────────────────────────────────────────────────────────────  

    public void render(DrawContext drawContext, float tickDelta) {  
        MinecraftClient mc = MinecraftClient.getInstance();  

        // Jangan render saat di menu  
        if (mc.currentScreen != null) return;  
        if (!stateMachine.isRunning() && lastError == null) return;  

        TextRenderer tr    = mc.textRenderer;  
        AutomationState state = stateMachine.getCurrentState();  

        // ── Kumpulkan baris HUD ───────────────────────────────────  
        List<HudLine> lines = new ArrayList<>();  

        // Header  
        lines.add(new HudLine("⚙ AutoCraft", COLOR_TITLE));  

        // State saat ini  
        lines.add(new HudLine(  
            "▶ " + formatState(state),  
            stateColor(state)  
        ));  

        // Job aktif  
        ctx.peekJob().ifPresent(job -> {  
            lines.add(new HudLine(  
                "  Item : " + shortKey(job.getItemKey()),  
                COLOR_JOB  
            ));  
            lines.add(new HudLine(  
                "  Amt  : " + job.getAmountNeeded(),  
                COLOR_JOB  
            ));  
        });  

        // Kedalaman stack  
        if (ctx.getDepth() > 0) {  
            lines.add(new HudLine(  
                "  Depth: " + ctx.getDepth() + " / 8",  
                COLOR_DEPTH  
            ));  
        }  

        // Error message (tampilkan 5 detik)  
        if (lastError != null) {  
            if (System.currentTimeMillis() - errorTimer < 5000) {  
                lines.add(new HudLine("✗ " + lastError, COLOR_STATE_ERROR));  
            } else {  
                lastError = null;  
            }  
        }  

        // ── Hitung dimensi background ─────────────────────────────  
        int maxWidth = lines.stream()  
            .mapToInt(l -> tr.getWidth(l.text()))  
            .max().orElse(0);  

        int bgW = maxWidth  + PADDING * 2;  
        int bgH = lines.size() * LINE_H + PADDING * 2;  

        // ── Render background ─────────────────────────────────────  
        drawContext.fill(  
            X_OFFSET,  
            Y_OFFSET,  
            X_OFFSET + bgW,  
            Y_OFFSET + bgH,  
            COLOR_BG  
        );  

        // ── Render teks ───────────────────────────────────────────  
        for (int i = 0; i < lines.size(); i++) {  
            HudLine line = lines.get(i);  
            drawContext.drawText(  
                tr,  
                line.text(),  
                X_OFFSET + PADDING,  
                Y_OFFSET + PADDING + i * LINE_H,  
                line.color(),  
                false  
            );  
        }  
    }  

    // ─────────────────────────────────────────────────────────────  
    // Helpers  
    // ─────────────────────────────────────────────────────────────  

    private String formatState(AutomationState state) {  
        return switch (state) {  
            case IDLE           -> "Idle";  
            case FINDING_GRID   -> "Mencari Grid...";  
            case GRID_OPEN      -> "Membuka Grid...";  
            case SCANNING       -> "Scanning Grid (hal. " + ctx.getGridPage() + ")";  
            case FINDING_MACHINE-> "Mencari Mesin...";  
            case MACHINE_OPEN   -> "Membuka Mesin...";  
            case CRAFTING       -> "Crafting...";  
            case VERIFYING      -> "Verifikasi Output...";  
            case DONE           -> "Selesai ✓";  
            case ERROR          -> "ERROR";  
        };  
    }  

    private int stateColor(AutomationState state) {  
        return switch (state) {  
            case ERROR          -> COLOR_STATE_ERROR;  
            case DONE, IDLE     -> COLOR_STATE_ACTIVE;  
            default             -> COLOR_STATE_WAIT;  
        };  
    }  

    private String shortKey(String itemKey) {  
        // "slimefun:COPPER_DUST" → "COPPER_DUST"  
        int colon = itemKey.lastIndexOf(':');  
        return colon >= 0 ? itemKey.substring(colon + 1) : itemKey;  
    }  

    // ── Record untuk baris HUD ────────────────────────────────────  
    private record HudLine(String text, int color) {}  
}

package com.example;  

import com.example.automation.*;  
import com.example.automation.hud.AutomationHudRenderer;  
import com.example.automation.impl.*;  
import net.fabricmc.api.ClientModInitializer;  
import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;  
import net.fabricmc.fabric.api.client.keybinding.v1.KeyBindingHelper;  
import net.fabricmc.fabric.api.client.rendering.v1.HudRenderCallback;  
import net.minecraft.client.option.KeyBinding;  
import net.minecraft.item.ItemStack;  
import org.lwjgl.glfw.GLFW;  
import org.slf4j.Logger;  
import org.slf4j.LoggerFactory;  

public class ExampleModClient implements ClientModInitializer {  

    private static final Logger LOGGER = LoggerFactory.getLogger("BapelMod");  

    // Key binding: Enter  
    private static KeyBinding triggerKey;  

    @Override  
    public void onInitializeClient() {  
        LOGGER.info("[BapelMod] Initializing automation engine...");  

        // ── 1. Build State Machine ────────────────────────────────  
        AutomationStateMachine stateMachine = AutomationStateMachine.builder()  
            .gridLocator(new NetworkGridLocatorImpl())  
            .machineLocator(new FastMachineLocatorImpl())  
            .subRecipeResolver(new SubRecipeResolverImpl())  
            .recipeLockExecutor(new RecipeLockExecutorImpl())  
            .inventoryChecker(new InventoryCheckerImpl())  
            .cameraController(new CameraControllerImpl())  
            .guiInteractor(new GuiInteractorImpl())  
            .gridScanner(new GridScannerImpl())  
            .build();  

        // ── 2. Expose context untuk HUD ───────────────────────────  
        AutomationContext ctx = stateMachine.getContext(); // tambahkan getter ke StateMachine  

        // ── 3. HUD Renderer ───────────────────────────────────────  
        AutomationHudRenderer hudRenderer =  
            new AutomationHudRenderer(stateMachine, ctx);  

        HudRenderCallback.EVENT.register(hudRenderer::render);  

        // ── 4. Tick engine setiap client tick ─────────────────────  
        ClientTickEvents.END_CLIENT_TICK.register(client ->  
            stateMachine.onTick()  
        );  

        // ── 5. Key Binding: Shift + Enter ─────────────────────────  
        triggerKey = KeyBindingHelper.registerKeyBinding(new KeyBinding(  
            "key.bapelmod.trigger",  
            GLFW.GLFW_KEY_ENTER,  
            "category.bapelmod"  
        ));  

        ClientTickEvents.END_CLIENT_TICK.register(client -> {  
            if (!triggerKey.wasPressed()) return;  
            if (client.currentScreen != null) return;  
            if (client.player == null) return;  

            // Cek Shift modifier  
            boolean isShiftHeld = client.options.sneakKey.isPressed();  
            if (!isShiftHeld) return;  

            // Ambil item di tangan utama sebagai target  
            ItemStack target = client.player.getMainHandStack();  
            if (target.isEmpty()) {  
                client.player.sendMessage(  
                    net.minecraft.text.Text.literal(  
                        "§e[BapelMod] Hold target item di tangan!"  
                    ), true  
                );  
                return;  
            }  

            // Buat initial job  
            String itemKey = resolveItemKey(target);  
            CraftingJob initialJob = new CraftingJob(itemKey, target, 1, 0);  

            LOGGER.info("[BapelMod] Starting chain for: {}", itemKey);  
            stateMachine.start(initialJob);  
        });  

        LOGGER.info("[BapelMod] Automation engine ready. Press Shift+Enter to start!");  
    }  

    // ─────────────────────────────────────────────────────────────  

    private String resolveItemKey(ItemStack stack) {  
        // Coba ambil Slimefun ID dari NBT  
        if (stack.hasNbt()) {  
            var nbt = stack.getNbt();  
            if (nbt != null && nbt.contains("PublicBukkitValues")) {  
                var sfTag = nbt.getCompound("PublicBukkitValues");  
                String sfId = sfTag.getString("slimefun:slimefun_item");  
                if (!sfId.isEmpty()) return "slimefun:" + sfId;  
            }  
        }  
        // Fallback ke vanilla item ID  
        return stack.getItem().toString();  
    }  
}
