package com.example.network;  

import com.example.automation.recipe.RecipeEntry;  
import com.example.automation.recipe.RecipeIngredient;  
import net.minecraft.network.PacketByteBuf;  
import net.minecraft.network.codec.PacketCodec;  
import net.minecraft.network.codec.PacketCodecs;  

import java.util.ArrayList;  
import java.util.List;  

/**  
 * Packet untuk sync satu RecipeEntry dari server ke client.  
 * Bisa digunakan untuk full sync (kirim banyak) atau delta (kirim satu).  
 *  
 * Wire format:  
 *   [String] outputKey  
 *   [int]    outputAmount  
 *   [String] requiredMachine  
 *   [int]    ingredientCount  
 *   repeat ingredientCount:  
 *     [String] ingredientKey  
 *     [int]    ingredientAmount  
 *     [bool]   isOptional  
 */  
public record RecipeSyncPacket(  
    String              outputKey,  
    int                 outputAmount,  
    String              requiredMachine,  
    List<IngredientData> ingredients  
) {  

    public record IngredientData(  
        String  itemKey,  
        int     amount,  
        boolean optional  
    ) {}  

    // ── Codec ─────────────────────────────────────────────────────  

    public static final PacketCodec<PacketByteBuf, RecipeSyncPacket> CODEC =  
        PacketCodec.of(RecipeSyncPacket::encode, RecipeSyncPacket::decode);  

    public static void encode(RecipeSyncPacket packet, PacketByteBuf buf) {  
        buf.writeString(packet.outputKey());  
        buf.writeInt(packet.outputAmount());  
        buf.writeString(packet.requiredMachine());  
        buf.writeInt(packet.ingredients().size());  
        for (IngredientData ing : packet.ingredients()) {  
            buf.writeString(ing.itemKey());  
            buf.writeInt(ing.amount());  
            buf.writeBoolean(ing.optional());  
        }  
    }  

    public static RecipeSyncPacket decode(PacketByteBuf buf) {  
        String outputKey       = buf.readString();  
        int    outputAmount    = buf.readInt();  
        String requiredMachine = buf.readString();  
        int    count           = buf.readInt();  

        List<IngredientData> ingredients = new ArrayList<>(count);  
        for (int i = 0; i < count; i++) {  
            ingredients.add(new IngredientData(  
                buf.readString(),  
                buf.readInt(),  
                buf.readBoolean()  
            ));  
        }  

        return new RecipeSyncPacket(outputKey, outputAmount, requiredMachine, ingredients);  
    }  

    // ── Convert to RecipeEntry ────────────────────────────────────  

    public RecipeEntry toRecipeEntry() {  
        List<RecipeIngredient> recipeIngredients = ingredients.stream()  
            .map(ing -> new RecipeIngredient(ing.itemKey(), ing.amount(), ing.optional()))  
            .toList();  

        return new RecipeEntry(  
            outputKey,  
            net.minecraft.item.ItemStack.EMPTY, // client tidak butuh real ItemStack  
            outputAmount,  
            recipeIngredients,  
            requiredMachine  
        );  
    }  
}

package com.example.automation.network;  

import com.example.automation.recipe.SlimefunRecipeRegistry;  
import com.example.automation.util.ChatFeedback;  
import com.example.network.RecipeSyncPacket;  
import net.fabricmc.fabric.api.client.networking.v1.ClientPlayNetworking;  
import net.minecraft.network.PacketByteBuf;  
import org.slf4j.Logger;  
import org.slf4j.LoggerFactory;  

/**  
 * Mendaftarkan semua custom packet handler di sisi client.  
 *  
 * Dipanggil sekali di onInitializeClient().  
 */  
public final class AutomationPacketHandler {  

    private static final Logger LOGGER = LoggerFactory.getLogger("AutomationPackets");  

    private AutomationPacketHandler() {}  

    // ─────────────────────────────────────────────────────────────  

    public static void register() {  
        registerRecipeSyncFull();  
        registerRecipeSyncDelta();  
        registerCraftConfirm();  
        registerMachineStatus();  

        LOGGER.info("[BapelMod] Custom packet handlers registered.");  
    }  

    // ─────────────────────────────────────────────────────────────  
    // Handlers  
    // ─────────────────────────────────────────────────────────────  

    /**  
     * Terima seluruh recipe registry dari server.  
     * Dipanggil saat player pertama kali join/respawn.  
     *  
     * Wire format:  
     *   [int]    totalRecipes  
     *   repeat:  RecipeSyncPacket per resep  
     */  
    private static void registerRecipeSyncFull() {  
        ClientPlayNetworking.registerGlobalReceiver(  
            PacketIds.RECIPE_SYNC_FULL,  
            (client, handler, buf, responseSender) -> {  

                // Baca semua packet di network thread  
                int totalRecipes = buf.readInt();  
                var packets = new java.util.ArrayList<RecipeSyncPacket>(totalRecipes);  
                for (int i = 0; i < totalRecipes; i++) {  
                    packets.add(RecipeSyncPacket.decode(buf));  
                }  

                // Execute di client/game thread  
                client.execute(() -> {  
                    SlimefunRecipeRegistry registry = SlimefunRecipeRegistry.get();  
                    registry.clear();  

                    for (RecipeSyncPacket pkt : packets) {  
                        registry.addRecipe(pkt.toRecipeEntry());  
                    }  

                    LOGGER.info("[BapelMod] Recipe sync complete: {} recipes loaded.",  
                        registry.totalRecipes());  
                    ChatFeedback.success("Recipe registry synced: "  
                        + registry.totalRecipes() + " recipes.");  
                });  
            }  
        );  
    }  

    /**  
     * Terima update delta (satu resep tambah/hapus).  
     *  
     * Wire format:  
     *   [byte]  action: 0=ADD, 1=REMOVE  
     *   RecipeSyncPacket (hanya jika ADD)  
     *   [String] outputKey (hanya jika REMOVE)  
     */  
    private static void registerRecipeSyncDelta() {  
        ClientPlayNetworking.registerGlobalReceiver(  
            PacketIds.RECIPE_SYNC_DELTA,  
            (client, handler, buf, responseSender) -> {  

                byte   action    = buf.readByte();  
                String outputKey = null;  
                RecipeSyncPacket packet = null;  

                if (action == 0) {       // ADD  
                    packet = RecipeSyncPacket.decode(buf);  
                } else if (action == 1) { // REMOVE  
                    outputKey = buf.readString();  
                }  

                final var finalPacket    = packet;  
                final var finalOutputKey = outputKey;  

                client.execute(() -> {  
                    SlimefunRecipeRegistry reg = SlimefunRecipeRegistry.get();  

                    if (action == 0 && finalPacket != null) {  
                        reg.addRecipe(finalPacket.toRecipeEntry());  
                        LOGGER.debug("Delta ADD: {}", finalPacket.outputKey());  

                    } else if (action == 1 && finalOutputKey != null) {  
                        reg.removeRecipe(finalOutputKey);  
                        LOGGER.debug("Delta REMOVE: {}", finalOutputKey);  
                    }  
                });  
            }  
        );  
    }  

    /**  
     * Server konfirmasi bahwa crafting berhasil.  
     *  
     * Wire format:  
     *   [String] itemKey yang selesai di-craft  
     *   [int]    jumlah output  
     */  
    private static void registerCraftConfirm() {  
        ClientPlayNetworking.registerGlobalReceiver(  
            PacketIds.CRAFT_CONFIRM,  
            (client, handler, buf, responseSender) -> {  

                String itemKey = buf.readString();  
                int    amount  = buf.readInt();  

                client.execute(() -> {  
                    LOGGER.info("[BapelMod] Server confirmed craft: {} x{}",  
                        itemKey, amount);  
                    // Event bus notification (opsional)  
                    com.example.automation.event.AutomationEventBus.get()  
                        .publish(  
                            com.example.automation.event.AutomationEvent.CRAFTING_VERIFIED,  
                            itemKey  
                        );  
                });  
            }  
        );  
    }  

    /**  
     * Server kirim status mesin (online/offline/busy).  
     *  
     * Wire format:  
     *   [int]  blockX  
     *   [int]  blockY  
     *   [int]  blockZ  
     *   [byte] status: 0=ONLINE, 1=OFFLINE, 2=BUSY  
     */  
    private static void registerMachineStatus() {  
        ClientPlayNetworking.registerGlobalReceiver(  
            PacketIds.MACHINE_STATUS,  
            (client, handler, buf, responseSender) -> {  

                int  x      = buf.readInt();  
                int  y      = buf.readInt();  
                int  z      = buf.readInt();  
                byte status = buf.readByte();  

                net.minecraft.util.math.BlockPos pos =  
                    new net.minecraft.util.math.BlockPos(x, y, z);  

                client.execute(() -> {  
                    MachineStatusCache.get().update(pos, MachineStatus.fromByte(status));  
                    LOGGER.debug("Machine status update: {} → {}", pos, status);  
                });  
            }  
        );  
    }  
}

package com.example.automation.network;  

import net.minecraft.util.math.BlockPos;  
import java.util.Map;  
import java.util.concurrent.ConcurrentHashMap;  

/**  
 * Cache status mesin yang dikirim server.  
 * Digunakan oleh FastMachineLocatorImpl untuk filter mesin yang available.  
 */  
public final class MachineStatusCache {  

    private static final MachineStatusCache INSTANCE = new MachineStatusCache();  
    public static MachineStatusCache get() { return INSTANCE; }  

    private final Map<BlockPos, MachineStatus> cache = new ConcurrentHashMap<>();  

    private MachineStatusCache() {}  

    public void update(BlockPos pos, MachineStatus status) {  
        if (status == MachineStatus.OFFLINE) {  
            cache.remove(pos);  
        } else {  
            cache.put(pos, status);  
        }  
    }  

    public MachineStatus getStatus(BlockPos pos) {  
        return cache.getOrDefault(pos, MachineStatus.ONLINE);  
    }  

    public boolean isAvailable(BlockPos pos) {  
        return getStatus(pos) == MachineStatus.ONLINE;  
    }  

    public void clear() { cache.clear(); }  
}

package com.example.automation.network;  

public enum MachineStatus {  
    ONLINE,   // siap dipakai  
    OFFLINE,  // tidak aktif  
    BUSY;     // sedang dipakai player lain  

    public static MachineStatus fromByte(byte b) {  
        return switch (b) {  
            case 1  -> OFFLINE;  
            case 2  -> BUSY;  
            default -> ONLINE;  
        };  
    }  
}

import net.minecraft.client.gui.screen.ingame.HandledScreen;

import org.spongepowered.asm.mixin.Mixin;

import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;  

/**  
 * Blokir input manual dari player saat automation sedang berjalan  
 * untuk mencegah konflik klik slot.  
 */  
@Mixin(HandledScreen.class)  
public class HandledScreenMixin {  

    @Inject(  
        method = "onMouseClick(Lnet/minecraft/screen/slot/Slot;IILnet/minecraft/screen/slot/SlotActionType;)V",  
        at = @At("HEAD"),  
        cancellable = true  
    )  
    private void onMouseClick(  
        net.minecraft.screen.slot.Slot slot,  
        int slotNum,  
        int button,  
        SlotActionType actionType,  
        CallbackInfo ci  
    ) {  
        // Jika automation sedang berjalan, blokir input manual player  
        if (ExampleModClient.AUTOMATION != null  
            && ExampleModClient.AUTOMATION.isRunning()  
        ) {  
            ci.cancel(); // blokir klik manual  
        }  
    }  
}

import org.spongepowered.asm.mixin.Mixin;  
import org.spongepowered.asm.mixin.injection.At;  
import org.spongepowered.asm.mixin.injection.Inject;  
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;  
import org.slf4j.Logger;  
import org.slf4j.LoggerFactory;  

@Mixin(ClientPlayNetworkHandler.class)  
public class ClientPlayNetworkHandlerMixin {  

    private static final Logger LOGGER = LoggerFactory.getLogger("BapelMixin");  

    /**  
     * Saat player join game:  
     * 1. Clear recipe registry lama  
     * 2. Clear machine status cache lama  
     * 3. Request recipe sync dari server  
     */  
    @Inject(  
        method = "onGameJoin",  
        at = @At("TAIL")  
    )  
    private void onGameJoin(GameJoinS2CPacket packet, CallbackInfo ci) {  
        LOGGER.info("[BapelMod] Player joined, requesting recipe sync...");  

        // Clear stale data  
        SlimefunRecipeRegistry.get().clear();  
        MachineStatusCache.get().clear();  

        // Kirim request sync ke server  
        if (ClientPlayNetworking.canSend(PacketIds.REQUEST_RECIPE_SYNC)) {  
            PacketByteBuf buf = new PacketByteBuf(Unpooled.buffer());  
            ClientPlayNetworking.send(PacketIds.REQUEST_RECIPE_SYNC, buf);  
            LOGGER.info("[BapelMod] Recipe sync request sent.");  
        } else {  
            LOGGER.warn("[BapelMod] Server tidak support packet sync. " +  
                "Menggunakan dev recipes.");  
        }  
    }  
}

package com.example.network;  

import com.example.automation.network.PacketIds;  
import com.example.automation.recipe.RecipeEntry;  
import com.example.automation.recipe.SlimefunRecipeRegistry;  
import net.fabricmc.fabric.api.networking.v1.ServerPlayNetworking;  
import net.minecraft.network.PacketByteBuf;  
import net.minecraft.server.network.ServerPlayerEntity;  
import io.netty.buffer.Unpooled;  
import org.slf4j.Logger;  
import org.slf4j.LoggerFactory;  

import java.util.List;  

/**  
 * Mengirim custom packet dari server ke client.  
 *  
 * Dipanggil dari:  
 *   - ServerPlayConnectionEvents.JOIN (kirim full sync)  
 *   - Saat resep berubah (kirim delta)  
 */  
public final class ServerPacketSender {  

    private static final Logger LOGGER = LoggerFactory.getLogger("ServerPacketSender");  

    private ServerPacketSender() {}  

    // ─────────────────────────────────────────────────────────────  

    /**  
     * Kirim full recipe registry ke satu player.  
     * Dipanggil saat player join atau REQUEST_RECIPE_SYNC diterima.  
     */  
    public static void sendFullRecipeSync(  
        ServerPlayerEntity player,  
        List<RecipeEntry>  recipes  
    ) {  
        if (!ServerPlayNetworking.canSend(player, PacketIds.RECIPE_SYNC_FULL)) {  
            LOGGER.warn("Player {} tidak support recipe sync packet", player.getName());  
            return;  
        }  

        PacketByteBuf buf = new PacketByteBuf(Unpooled.buffer());  
        buf.writeInt(recipes.size());  

        for (RecipeEntry entry : recipes) {  
            RecipeSyncPacket pkt = toPacket(entry);  
            RecipeSyncPacket.encode(pkt, buf);  
        }  

        ServerPlayNetworking.send(player, PacketIds.RECIPE_SYNC_FULL, buf);  
        LOGGER.info("Full recipe sync sent to {}: {} recipes",  
            player.getName().getString(), recipes.size());  
    }  

    /**  
     * Broadcast delta update (satu resep baru) ke semua player online.  
     */  
    public static void broadcastRecipeAdd(  
        net.minecraft.server.MinecraftServer server,  
        RecipeEntry newRecipe  
    ) {  
        PacketByteBuf buf = new PacketByteBuf(Unpooled.buffer());  
        buf.writeByte(0); // ADD  
        RecipeSyncPacket.encode(toPacket(newRecipe), buf);  

        for (ServerPlayerEntity player : server.getPlayerManager().getPlayerList()) {  
            if (ServerPlayNetworking.canSend(player, PacketIds.RECIPE_SYNC_DELTA)) {  
                ServerPlayNetworking.send(player, PacketIds.RECIPE_SYNC_DELTA, buf.copy());  
            }  
        }  

        LOGGER.info("Recipe delta ADD broadcast: {}", newRecipe.outputKey());  
    }  

    /**  
     * Broadcast delta update (hapus resep) ke semua player online.  
     */  
    public static void broadcastRecipeRemove(  
        net.minecraft.server.MinecraftServer server,  
        String outputKey  
    ) {  
        PacketByteBuf buf = new PacketByteBuf(Unpooled.buffer());  
        buf.writeByte(1); // REMOVE  
        buf.writeString(outputKey);  

        for (ServerPlayerEntity player : server.getPlayerManager().getPlayerList()) {  
            if (ServerPlayNetworking.canSend(player, PacketIds.RECIPE_SYNC_DELTA)) {  
                ServerPlayNetworking.send(player, PacketIds.RECIPE_SYNC_DELTA, buf.copy());  
            }  
        }  

        LOGGER.info("Recipe delta REMOVE broadcast: {}", outputKey);  
    }  

    /**  
     * Kirim konfirmasi crafting ke player tertentu.  
     */  
    public static void sendCraftConfirm(  
        ServerPlayerEntity player,  
        String itemKey,  
        int    amount  
    ) {  
        PacketByteBuf buf = new PacketByteBuf(Unpooled.buffer());  
        buf.writeString(itemKey);  
        buf.writeInt(amount);  
        ServerPlayNetworking.send(player, PacketIds.CRAFT_CONFIRM, buf);  
    }  

    // ─────────────────────────────────────────────────────────────  

    private static RecipeSyncPacket toPacket(RecipeEntry entry) {  
        var ingredients = entry.ingredients().stream()  
            .map(ing -> new RecipeSyncPacket.IngredientData(  
                ing.itemKey(), ing.amount(), ing.isOptional()  
            ))  
            .toList();  

        return new RecipeSyncPacket(  
            entry.outputKey(),  
            entry.outputAmount(),  
            entry.requiredMachine(),  
            ingredients  
        );  
    }  
}

package com.example.automation;  

import com.example.automation.state.AutomationState;  
import org.junit.jupiter.api.*;  
import org.mockito.Mock;  
import org.mockito.MockitoAnnotations;  

import static org.assertj.core.api.Assertions.*;  
import static org.mockito.Mockito.*;  

@DisplayName("AutomationStateMachine Tests")  
class AutomationStateMachineTest {  

    @Mock private com.example.automation.infra.NetworkGridLocator  gridLocator;  
    @Mock private com.example.automation.infra.FastMachineLocator  machineLocator;  
    @Mock private com.example.automation.infra.SubRecipeResolver   subRecipeResolver;  
    @Mock private com.example.automation.infra.RecipeLockExecutor  recipeLockExecutor;  
    @Mock private com.example.automation.infra.InventoryChecker    inventoryChecker;  
    @Mock private com.example.automation.infra.CameraController    cameraController;  
    @Mock private com.example.automation.infra.GuiInteractor       guiInteractor;  
    @Mock private com.example.automation.infra.GridScanner         gridScanner;  

    private AutomationStateMachine stateMachine;  
    private AutoCloseable          mocks;  

    @BeforeEach  
    void setUp() {  
        mocks = MockitoAnnotations.openMocks(this);  

        stateMachine = AutomationStateMachine.builder()  
            .gridLocator(gridLocator)  
            .machineLocator(machineLocator)  
            .subRecipeResolver(subRecipeResolver)  
            .recipeLockExecutor(recipeLockExecutor)  
            .inventoryChecker(inventoryChecker)  
            .cameraController(cameraController)  
            .guiInteractor(guiInteractor)  
            .gridScanner(gridScanner)  
            .build();  
    }  

    @AfterEach  
    void tearDown() throws Exception {  
        mocks.close();  
    }  

    // ─────────────────────────────────────────────────────────────  

    @Test  
    @DisplayName("Initial state harus IDLE")  
    void shouldStartInIdleState() {  
        assertThat(stateMachine.getCurrentState())  
            .isEqualTo(AutomationState.IDLE);  
        assertThat(stateMachine.isRunning()).isFalse();  
    }  

    @Test  
    @DisplayName("start() harus pindah ke FINDING_GRID")  
    void shouldTransitionToFindingGridOnStart() {  
        CraftingJob job = makeJob("slimefun:COPPER_INGOT", 1);  
        stateMachine.start(job);  

        assertThat(stateMachine.getCurrentState())  
            .isEqualTo(AutomationState.FINDING_GRID);  
        assertThat(stateMachine.isRunning()).isTrue();  
    }  

    @Test  
    @DisplayName("cancelChain() harus kembali ke IDLE")  
    void shouldReturnToIdleOnCancel() {  
        stateMachine.start(makeJob("slimefun:COPPER_INGOT", 1));  
        stateMachine.cancelChain();  

        assertThat(stateMachine.getCurrentState())  
            .isEqualTo(AutomationState.IDLE);  
        assertThat(stateMachine.isRunning()).isFalse();  
    }  

    @Test  
    @DisplayName("start() saat running harus cancel chain lama")  
    void shouldCancelOldChainBeforeStartingNew() {  
        CraftingJob job1 = makeJob("slimefun:COPPER_INGOT", 1);  
        CraftingJob job2 = makeJob("slimefun:IRON_INGOT", 1);  

        stateMachine.start(job1);  
        assertThat(stateMachine.isRunning()).isTrue();  

        stateMachine.start(job2); // trigger baru saat running  
        assertThat(stateMachine.isRunning()).isTrue();  
        assertThat(stateMachine.getCurrentState())  
            .isEqualTo(AutomationState.FINDING_GRID);  
    }  

    @Test  
    @DisplayName("Transisi ilegal harus diabaikan")  
    void shouldIgnoreIllegalTransitions() {  
        // IDLE → CRAFTING adalah ilegal  
        assertThat(AutomationState.CRAFTING.canTransitionFrom(AutomationState.IDLE))  
            .isFalse();  
    }  

    @Test  
    @DisplayName("ERROR selalu bisa dicapai dari state manapun")  
    void errorShouldBeReachableFromAnyState() {  
        for (AutomationState state : AutomationState.values()) {  
            assertThat(AutomationState.ERROR.canTransitionFrom(state)).isTrue();  
        }  
    }  

    // ─────────────────────────────────────────────────────────────  

    private CraftingJob makeJob(String key, int amount) {

package com.example.automation;  

import com.example.automation.recipe.*;  
import org.junit.jupiter.api.*;  

import java.util.List;  

import static org.assertj.core.api.Assertions.*;  

@DisplayName("Circular Recipe Guard Tests")  
class CircularRecipeGuardTest {  

    private AutomationContext ctx;  

    @BeforeEach  
    void setUp() {  
        ctx = new AutomationContext();  
    }  

    // ─────────────────────────────────────────────────────────────  

    @Test  
    @DisplayName("pushJob kedua kali dengan key sama harus ditolak (circular)")  
    void shouldRejectCircularRecipe() {  
        CraftingJob jobA = new CraftingJob(  
            "slimefun:ITEM_A", net.minecraft.item.ItemStack.EMPTY, 1, 0  
        );  
        CraftingJob jobACopy = new CraftingJob(  
            "slimefun:ITEM_A", net.minecraft.item.ItemStack.EMPTY, 1, 1  
        );  

        boolean first  = ctx.pushJob(jobA);  
        boolean second = ctx.pushJob(jobACopy); // circular!  

        assertThat(first).isTrue();  
        assertThat(second).isFalse(); // harus ditolak  
    }  

    @Test  
    @DisplayName("pushJob berbeda key harus diterima")  
    void shouldAcceptDifferentKeys() {  
        CraftingJob jobA = new CraftingJob(  
            "slimefun:ITEM_A", net.minecraft.item.ItemStack.EMPTY, 1, 0  
        );  
        CraftingJob jobB = new CraftingJob(  
            "slimefun:ITEM_B", net.minecraft.item.ItemStack.EMPTY, 1, 1  
        );  

        assertThat(ctx.pushJob(jobA)).isTrue();  
        assertThat(ctx.pushJob(jobB)).isTrue();  
    }  

    @Test  
    @DisplayName("popJob harus release circular guard untuk key tersebut")  
    void shouldReleaseGuardOnPop() {  
        CraftingJob job = new CraftingJob(  
            "slimefun:ITEM_A", net.minecraft.item.ItemStack.EMPTY, 1, 0  
        );  

        ctx.pushJob(job);  
        ctx.popJob(); // release  

        // Sekarang bisa push lagi  
        CraftingJob jobAgain = new CraftingJob(  
            "slimefun:ITEM_A", net.minecraft.item.ItemStack.EMPTY, 1, 0  
        );  
        assertThat(ctx.pushJob(jobAgain)).isTrue();  
    }  

    @Test  
    @DisplayName("stack depth harus dibatasi MAX_DEPTH = 8")  
    void shouldEnforceMaxDepth() {  
        for (int i = 0; i < 8; i++) {  
            CraftingJob job = new CraftingJob(  
                "slimefun:ITEM_" + i,  
                net.minecraft.item.ItemStack.EMPTY, 1, i  
            );  
            assertThat(ctx.pushJob(job)).isTrue();  
        }  

        // Push ke-9 harus gagal  
        CraftingJob overflow = new CraftingJob(  
            "slimefun:ITEM_OVERFLOW",  
            net.minecraft.item.ItemStack.EMPTY, 1, 8  
        );  
        assertThat(ctx.pushJob(overflow)).isFalse();  
    }  
}
